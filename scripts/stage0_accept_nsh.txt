# Stage0 板端验收（MThings 从站仿真已开，COM6=9600 8N1，勿开采集）
# 在 COM3 NSH 逐段粘贴执行；每段等 nsh> 回来再贴下一段。
# 固件：bash scripts/build.sh emmc && bash scripts/flash.sh（READ_TO=2000ms）

# --- §1 eMMC ---
ls /dev/mmcsd0
mount
echo stage0-final > /mnt/emmc/vg_ac.txt
cat /mnt/emmc/vg_ac.txt

# --- §2 RS485（物理层，MThings 可不停从站监听）---
vgrs485 tx

# --- §4a 帧统计 inject（不写总线）---
vgstats inject 1 crc
vgstats inject 1 timeout
vgstats dump 1

# --- §3 + §4b 从站 1（温湿度 reg0..1）---
vgstats reset 1
vgmodbus -a 1 -r 0 -c 2 -n 5 -i 1
vgstats dump 1

# --- §3 + §4b 从站 2（水浸 reg2）---
vgstats reset 2
vgmodbus -a 2 -r 2 -c 1 -n 5 -i 1
vgstats dump 2

# --- §3 + §4b 从站 3（PT100 reg0..3）---
vgstats reset 3
vgmodbus -a 3 -r 0 -c 4 -n 5 -i 1
vgstats dump 3

# --- §5 掉电安全（抽样，非 50×）---
vgcfg probe
vgcfg commit stage0ac
vgcfg dump
